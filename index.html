/** * GEEZ Serial system - verify + admin (shaft search, NFC, photos) + alerts + mapping * * TABS in your Google Sheet: * - First tab = products / verification. Headers (row 1, any order): * shaft_number nfc_number box_number * (extra columns temp_box_number, proof_shaft, proof_nfc, proof_box are * created automatically - don't worry about them.) * - A tab named "shaft number" = your master shaft list (numbers in column A). * * SET THESE: */ var ADD_KEY = "CHANGE-THIS-PASSWORD"; // your admin password var OWNER_EMAIL = "geeztech.info@gmail.com"; // where login alerts go var SHAFT_SHEET = "shaft number"; // tab holding the master shaft list var DRIVE_FOLDER = "GEEZ proofs"; // Drive folder for proof photos var DATA_SHEET = ""; // "" = first tab (products / verification) var MAX_LEN = 80; var MAX_TRIES_PER_MIN = 30; var MAX_FAILS = 5; var LOCK_MINUTES = 15; function doGet(e) { return route(e); } function doPost(e) { return route(e); } function route(e) { var p = (e && e.parameter) ? e.parameter : {}; if (p.action === "login") return login(p); if (p.action === "shafts") return shafts(p); if (p.action === "add") return addProduct(p); return verify(p); } /* ---------- sheet menu tool: temp -> real box numbers ---------- */ function onOpen() { SpreadsheetApp.getUi() .createMenu("GEEZ Tools") .addItem("Replace temp box numbers with real", "syncRealNumbers") .addToUi(); } function syncRealNumbers() { var ui = SpreadsheetApp.getUi(); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var main = getDataSheet_(); var map = ss.getSheetByName("mapping"); if (!map) { ui.alert('Make a tab named "mapping" with headers temp_number and real_number, then paste the factory list.'); return; } var mrows = map.getDataRange().getValues(); var mhead = mrows[0].map(function (x) { return String(x).trim().toLowerCase(); }); var ti = mhead.indexOf("temp_number"), ri = mhead.indexOf("real_number"); if (ti === -1 || ri === -1) { ui.alert('The "mapping" tab needs headers: temp_number and real_number.'); return; } var dict = {}; for (var m = 1; m < mrows.length; m++) { var t = clean(mrows[m][ti]); if (t && mrows[m][ri] !== "") dict[t] = String(mrows[m][ri]).trim(); } var data = main.getDataRange().getValues(); if (data.length < 2) { ui.alert("No products found."); return; } var head = data[0].map(function (x) { return String(x).trim().toLowerCase(); }); var boxCol = head.indexOf("box_number"); if (boxCol === -1) { ui.alert('No "box_number" column found.'); return; } var tempCol = head.indexOf("temp_box_number"); if (tempCol === -1) { tempCol = data[0].length; main.getRange(1, tempCol + 1).setValue("temp_box_number"); } var replaced = 0, notFound = 0; for (var r = 1; r < data.length; r++) { var cur = clean(data[r][boxCol]); if (!cur) continue; if (dict.hasOwnProperty(cur)) { main.getRange(r + 1, tempCol + 1).setValue(data[r][boxCol]); main.getRange(r + 1, boxCol + 1).setValue(dict[cur]); replaced++; } else { notFound++; } } ui.alert("Done.\nReplaced: " + replaced + "\nNot found: " + notFound + "\nOld temps saved in temp_box_number."); } catch (err) { ui.alert("Error: " + err); } } /* ---------- admin login ---------- */ function login(p) { try { if (isLocked_()) return json_({ ok: false, error: "locked" }); if (String(p.key || "") !== ADD_KEY) { var f = bumpFails_(); notify_("GEEZ admin: FAILED login", "Wrong password at " + now_() + ". Attempts: " + f); return json_({ ok: false, error: "wrong_password" }); } resetFails_(); notify_("GEEZ admin: login", "Login at " + now_() + ". If this was not you, change your password."); return json_({ ok: true }); } catch (err) { return json_({ ok: false, error: "server" }); } } /* ---------- shaft list for autocomplete ---------- */ function shafts(p) { try { if (String(p.key || "") !== ADD_KEY) return json_({ ok: false, error: "wrong_password" }); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = findSheet_(ss, SHAFT_SHEET); if (!sh) return json_({ ok: true, shafts: [] }); var vals = sh.getDataRange().getValues(); var list = []; for (var r = 0; r < vals.length; r++) { var v = String(vals[r][0] == null ? "" : vals[r][0]).trim(); if (!v) continue; if (r === 0 && v.toLowerCase().indexOf("shaft") !== -1) continue; // skip header list.push(v); } return json_({ ok: true, shafts: list }); } catch (err) { return json_({ ok: false, error: "server" }); } } /* ---------- customer verify ---------- */ function verify(p) { var out = { valid: false }; try { if (!allowRequest_()) return json_({ valid: false, error: "busy" }); var shaft = clean(p.shaft), nfc = clean(p.nfc), box = clean(p.box); if (!shaft || !nfc || !box) return json_(out); var sheet = getDataSheet_(); if (!sheet) return json_(out); var data = sheet.getDataRange().getValues(); if (data.length < 2) return json_(out); var idx = headerIndexes_(data[0]); if (idx.shaft === -1 || idx.nfc === -1 || idx.box === -1) return json_(out); for (var r = 1; r < data.length; r++) { if (clean(data[r][idx.shaft]) === shaft && clean(data[r][idx.nfc]) === nfc && clean(data[r][idx.box]) === box) { out.valid = true; break; } } } catch (err) { out = { valid: false }; } return json_(out); } /* ---------- admin add (with optional photos) ---------- */ function addProduct(p) { try { if (isLocked_()) return json_({ ok: false, error: "locked" }); if (String(p.key || "") !== ADD_KEY) { bumpFails_(); return json_({ ok: false, error: "wrong_password" }); } var shaft = clean(p.shaft), nfc = clean(p.nfc), box = clean(p.box); if (!shaft || !nfc || !box) return json_({ ok: false, error: "missing_fields" }); if (shaft.length > MAX_LEN || nfc.length > MAX_LEN || box.length > MAX_LEN) return json_({ ok: false, error: "too_long" }); var sheet = getDataSheet_(); if (!sheet) return json_({ ok: false, error: "no_sheet" }); var data = sheet.getDataRange().getValues(); if (data.length === 0 || (data.length === 1 && data[0].join("") === "")) { sheet.getRange(1, 1, 1, 3).setValues([["shaft_number", "nfc_number", "box_number"]]); data = sheet.getDataRange().getValues(); } var idx = headerIndexes_(data[0]); if (idx.shaft === -1 || idx.nfc === -1 || idx.box === -1) return json_({ ok: false, error: "bad_headers" }); for (var r = 1; r < data.length; r++) { if (clean(data[r][idx.box]) === box) return json_({ ok: false, error: "duplicate_box" }); } // upload optional photos, get links var linkShaft = p.pshaft ? savePhoto_(p.pshaft, shaft + "_shaft") : ""; var linkNfc = p.pnfc ? savePhoto_(p.pnfc, shaft + "_nfc") : ""; var linkBox = p.pbox ? savePhoto_(p.pbox, shaft + "_box") : ""; var cShaftPhoto = ensureCol_(sheet, "proof_shaft"); var cNfcPhoto = ensureCol_(sheet, "proof_nfc"); var cBoxPhoto = ensureCol_(sheet, "proof_box"); var width = sheet.getLastColumn(); var row = new Array(width).fill(""); row[idx.shaft] = shaft; row[idx.nfc] = nfc; row[idx.box] = box; row[cShaftPhoto] = linkShaft; row[cNfcPhoto] = linkNfc; row[cBoxPhoto] = linkBox; sheet.appendRow(row); return json_({ ok: true }); } catch (err) { return json_({ ok: false, error: "server" }); } } /* ---------- helpers ---------- */ function getDataSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return DATA_SHEET ? ss.getSheetByName(DATA_SHEET) : ss.getSheets()[0]; } function findSheet_(ss, name) { var s = ss.getSheetByName(name); if (s) return s; var all = ss.getSheets(); for (var i = 0; i < all.length; i++) { if (all[i].getName().trim().toLowerCase() === String(name).trim().toLowerCase()) return all[i]; } return null; } function headerIndexes_(headerRow) { var h = headerRow.map(function (x) { return String(x).trim().toLowerCase(); }); return { shaft: h.indexOf("shaft_number"), nfc: h.indexOf("nfc_number"), box: h.indexOf("box_number") }; } function ensureCol_(sheet, name) { var head = sheet.getRange(1, 1, 1, Math.max(1, sheet.getLastColumn())).getValues()[0] .map(function (x) { return String(x).trim().toLowerCase(); }); var i = head.indexOf(name.toLowerCase()); if (i !== -1) return i; var col = sheet.getLastColumn(); sheet.getRange(1, col + 1).setValue(name); return col; } function savePhoto_(dataUrl, name) { try { var m = String(dataUrl).match(/^data:([^;]+);base64,(.*)$/); if (!m) return ""; var blob = Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], name + ".jpg"); var folder = getFolder_(); var file = folder.createFile(blob); return file.getUrl(); } catch (e) { return ""; } } function getFolder_() { var it = DriveApp.getFoldersByName(DRIVE_FOLDER); return it.hasNext() ? it.next() : DriveApp.createFolder(DRIVE_FOLDER); } function clean(v) { if (v === null || v === undefined) return ""; return String(v).toUpperCase().replace(/\s+/g, ""); } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function now_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss"); } function notify_(s, b) { try { if (OWNER_EMAIL) MailApp.sendEmail(OWNER_EMAIL, s, b); } catch (e) {} } function allowRequest_() { try { var c = CacheService.getScriptCache(), k = "rl_" + Math.floor(Date.now() / 60000); var n = parseInt(c.get(k) || "0", 10); if (n >= MAX_TRIES_PER_MIN) return false; c.put(k, String(n + 1), 90); return true; } catch (e) { return true; } } function bumpFails_() { var c = CacheService.getScriptCache(), n = parseInt(c.get("fails") || "0", 10) + 1; c.put("fails", String(n), LOCK_MINUTES * 60); if (n >= MAX_FAILS) c.put("locked", "1", LOCK_MINUTES * 60); return n; } function resetFails_() { var c = CacheService.getScriptCache(); c.remove("fails"); c.remove("locked"); } function isLocked_() { try { return CacheService.getScriptCache().get("locked") === "1"; } catch (e) { return false; } }
